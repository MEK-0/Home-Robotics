cmake_minimum_required(VERSION 3.8)
project(home_robotics_manipulation)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
find_package(controller_manager_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(shape_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(yaml_cpp_vendor REQUIRED)
add_library(object_registry src/object_registry.cpp)
target_include_directories(object_registry PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(object_registry geometry_msgs shape_msgs moveit_msgs yaml_cpp_vendor)
target_link_libraries(object_registry yaml-cpp)
add_executable(dynamic_object_scene_sync src/dynamic_object_scene_sync.cpp)
ament_target_dependencies(dynamic_object_scene_sync ament_index_cpp rclcpp moveit_msgs std_srvs)
target_link_libraries(dynamic_object_scene_sync object_registry)
add_executable(cube_collision_validation src/cube_collision_validation.cpp)
ament_target_dependencies(cube_collision_validation moveit_ros_planning_interface moveit_msgs rclcpp sensor_msgs)
add_library(grasp_pose_generator src/grasp_pose_generator.cpp)
target_include_directories(grasp_pose_generator PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(grasp_pose_generator moveit_ros_planning_interface)
target_link_libraries(grasp_pose_generator object_registry)
add_executable(grasp_pose_validation src/grasp_pose_validation.cpp)
ament_target_dependencies(grasp_pose_validation ament_index_cpp moveit_ros_planning_interface moveit_msgs rclcpp sensor_msgs visualization_msgs)
target_link_libraries(grasp_pose_validation grasp_pose_generator)
add_executable(cube_pick_lift_return_demo src/cube_pick_lift_return_demo.cpp)
ament_target_dependencies(cube_pick_lift_return_demo ament_index_cpp moveit_ros_planning_interface moveit_msgs rclcpp rclcpp_action sensor_msgs visualization_msgs control_msgs controller_manager_msgs std_msgs std_srvs)
target_link_libraries(cube_pick_lift_return_demo grasp_pose_generator)
add_executable(cube_pick_place_demo src/cube_pick_lift_return_demo.cpp)
target_compile_definitions(cube_pick_place_demo PRIVATE PHASE4_PICK_PLACE=1)
ament_target_dependencies(cube_pick_place_demo ament_index_cpp moveit_ros_planning_interface moveit_msgs rclcpp rclcpp_action sensor_msgs visualization_msgs control_msgs controller_manager_msgs std_msgs std_srvs)
target_link_libraries(cube_pick_place_demo grasp_pose_generator)
install(TARGETS cube_pick_place_demo cube_pick_lift_return_demo grasp_pose_generator grasp_pose_validation object_registry dynamic_object_scene_sync cube_collision_validation
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(PROGRAMS scripts/benchmark_pick_place.py scripts/validate_dynamic_scene.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_placement test/test_placement.cpp)
  ament_target_dependencies(test_placement ament_index_cpp)
  target_link_libraries(test_placement object_registry grasp_pose_generator)
  ament_add_gtest(test_cube_demo test/test_cube_demo.cpp)
  ament_target_dependencies(test_cube_demo rclcpp rclcpp_action control_msgs moveit_msgs)
  target_include_directories(test_cube_demo PRIVATE include)
  ament_add_gtest(test_grasp_pose test/test_grasp_pose.cpp)
  ament_target_dependencies(test_grasp_pose ament_index_cpp)
  target_link_libraries(test_grasp_pose grasp_pose_generator)
  ament_add_gtest(test_registry test/test_registry.cpp)
  ament_target_dependencies(test_registry ament_index_cpp)
  target_link_libraries(test_registry object_registry)
endif()
ament_export_include_directories(include)
ament_export_libraries(object_registry grasp_pose_generator)
ament_export_dependencies(geometry_msgs shape_msgs moveit_msgs yaml_cpp_vendor moveit_ros_planning_interface)
ament_package()

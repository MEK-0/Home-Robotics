cmake_minimum_required(VERSION 3.8)
project(home_robotics_task_executor)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(controller_manager_msgs REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(yaml_cpp_vendor REQUIRED)
find_package(home_robotics_interfaces REQUIRED)
find_package(home_robotics_manipulation REQUIRED)
add_library(task_executor_core src/task_executor.cpp src/manipulation_adapter.cpp)
target_include_directories(task_executor_core PUBLIC include)
ament_target_dependencies(task_executor_core ament_index_cpp rclcpp rclcpp_action geometry_msgs controller_manager_msgs moveit_msgs yaml_cpp_vendor home_robotics_interfaces home_robotics_manipulation)
target_link_libraries(task_executor_core yaml-cpp)
add_executable(task_executor src/main.cpp)
target_link_libraries(task_executor task_executor_core)
install(TARGETS task_executor DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_task_contract test/test_task_contract.cpp)
  ament_target_dependencies(test_task_contract ament_index_cpp home_robotics_interfaces)
  target_include_directories(test_task_contract PRIVATE include)
  target_link_libraries(test_task_contract yaml-cpp)
  ament_add_gtest(test_task_executor_action test/test_task_executor_action.cpp)
  ament_target_dependencies(test_task_executor_action rclcpp rclcpp_action home_robotics_interfaces)
  target_include_directories(test_task_executor_action PRIVATE include)
  target_link_libraries(test_task_executor_action task_executor_core)
endif()

ament_package()
